class Solution {
public:
    int longestOnes(vector<int>& nums, int k) {
        int WS = 0;
        int zeroCnt = 0;
        int left = 0;
        for(int right = 0;right<nums.size();right++){
            if(nums[right] == 0)
            zeroCnt++;

            while(zeroCnt>k){
                if(nums[left] == 0)
                zeroCnt--;

                left++;
            }

            WS = max(WS, right-left+1);

        }
        return WS;
    }
};
