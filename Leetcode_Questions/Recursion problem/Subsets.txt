class Solution {
public:

    void subset(vector<int> &nums, int i, vector<int> &curr, vector<vector<int>> &ds){
       if(i == nums.size()){
       ds.push_back(curr);
       return;
       }

        subset(nums, i+1, curr, ds);
        curr.push_back(nums[i]);

        subset(nums, i+1, curr, ds);
        curr.pop_back();
    }
    vector<vector<int>> subsets(vector<int>& nums) {
        int n = nums.size();
        vector<vector<int>> ds;
        vector<int> curr;
        subset(nums, 0, curr, ds);

        return ds;
    }
};
