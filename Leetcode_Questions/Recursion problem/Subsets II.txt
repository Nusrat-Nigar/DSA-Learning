class Solution {
public:
    void f(vector<int> &nums, int i, int n, vector<int> &curr, vector<vector<int>> &ds){
        if(i == n){
            ds.push_back(curr);
            return;
        }
        curr.push_back(nums[i]);
        f(nums, i+1, n, curr, ds);
        curr.pop_back();

        while(i+1<n  && nums[i] == nums[i+1])
        i++;

        f(nums, i+1, n, curr, ds );
    }


    vector<vector<int>> subsetsWithDup(vector<int>& nums) {
        int n = nums.size();
        vector<int> curr;
        vector<vector<int>> ds;
        sort(nums.begin(), nums.end());
        f(nums, 0, n, curr, ds);
        return ds;
    }
};
