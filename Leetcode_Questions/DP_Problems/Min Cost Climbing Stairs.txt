class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) {
        int n = cost.size();
        vector<int>dp(n);
        for(int i=0;i<n;i++){
            if(i<2)
             dp[i] = cost[i];

            else
            dp[i] = cost[i]+min(dp[i-1], dp[i-2]);
        }
        return min(dp[n-1], dp[n-2]);
    
    }
};



class Solution {
public:
    int minCostClimbingStairs(vector<int>& cost) {
       int n = cost.size();
       int prev1 = cost[0];
       int prev2 = cost[1];
      if(n<2)
      return min(prev1, prev2);

      for(int i=2;i<n;i++){
          int curr = cost[i]+min(prev1, prev2);
          prev1 = prev2;
          prev2 = curr;      
          }

          return min(prev1, prev2);
      
    
    }
};
