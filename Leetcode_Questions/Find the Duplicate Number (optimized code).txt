// 1st approach

class Solution {
public:
    int findDuplicate(vector<int>& nums) {
     int n = nums.size();

     for(int i=0;i<n;i++)
     {
      
         int index = abs(nums[i]);
         if(nums[index]<0)
         {
             return index;
         }

         nums[index] *= -1;
     }
     return 0;
     }
}



// 2nd approach

class Solution {
public:
    int findDuplicate(vector<int>& nums) {

    while(nums[0] != nums[nums[0]]){
        swap(nums[0], nums[nums[0]]);
    }

    return nums[0];
    }
};
