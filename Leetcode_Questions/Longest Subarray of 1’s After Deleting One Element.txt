class Solution {
public:
    int longestSubarray(vector<int>& nums) {
        int WS = 0;
        int left=0;
        int cnt=0;
        for(int right=0;right<nums.size();right++){
            if(nums[right] == 0){
                cnt++;
            }

            while(cnt>1){
                if(nums[left] ==0){
                    cnt--;
                }
                left++;
            }

            WS = max(WS, right-left+1-cnt);
        }
        return WS == nums.size()? WS-1:WS;
    }
};
