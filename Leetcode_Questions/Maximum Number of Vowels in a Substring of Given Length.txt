class Solution {
public:
    int maxVowels(string s, int k) {
        string vowels = "aeiou";
        int WS = 0;
        int count=0;
        for(int i=0;i<k;i++){
            if(vowels.find(s[i]) != string::npos)
            count++;

            WS += s[i];
        }

           int maxCnt = count;
         for(int i=k;i<s.size();i++){
            if(vowels.find(s[i-k]) != string::npos)
            count--;

            if(vowels.find(s[i]) != string::npos)
            count++;

            WS += s[i]-s[i-k];

           maxCnt = max(maxCnt, count);

         }

         return maxCnt;
    }
};
